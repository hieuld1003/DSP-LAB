// --- Exercise 1.2 ---
clf();

// 1. Continuous-time signal xa(t) (5 periods: 0 to 0.1s)
F = 50;
T = 1 / F;
t = 0 : 0.0001 : 5 * T;
xa = 3 * sin(100 * %pi * t);

// 2 & 3. Discrete-time signal x(n) (5 periods, N=6 -> 30 samples)
Fs = 300;
n = 0:29;
xn = 3 * sin((%pi / 3) * n);

// 4. Quantized signal xq(n) using truncation method
delta = 0.1;
xq = floor(xn / delta) * delta;

// 5. Unified 3-panel display
subplot(3, 1, 1);
plot(t, xa, 'b-');
xlabel("Time t (seconds)"); ylabel("Amplitude");
title("1. Analog Signal: x_a(t) = 3*sin(100*%pi*t) (5 periods)");
xgrid(1);

subplot(3, 1, 2);
plot2d3('gnn', n, xn); plot(n, xn, 'ro');
xlabel("Sample index n"); ylabel("Amplitude");
title("2. Discrete-Time Signal: x(n) = 3*sin(\pi/3 * n) (5 periods, N=6, 30 samples)");
xgrid(1);

subplot(3, 1, 3);
plot2d3('gnn', n, xq); plot(n, xq, 'mo');
xlabel("Sample index n"); ylabel("Amplitude");
title("3. Quantized Signal: x_q(n) with \Delta = 0.1 (Truncation method)");
xgrid(1);

// 1. Define time index and input sequence: x(n) = {1, 3 (origin), -2}

// 2. Compute time-reversed signal x(-n)

// 3. Compute even and odd components

title("Signal x(n)");

ylabel("x(n)");

title("Even Component x_e(n)");

ylabel("x_e(n)");
