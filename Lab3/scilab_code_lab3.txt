// ----------------------------------------------------------------------------
// Exercise 1: Delay Operation y(n) = x(n - k), k > 0
// ----------------------------------------------------------------------------
function [yn, yorigin] = delay(xn, xorigin, k)
    yn = xn;
    yorigin = xorigin - k;
    
    // Determine discrete time axes
    nx = (1:length(xn)) - xorigin;
    ny = (1:length(yn)) - yorigin;
    
    // Graphical display in a single figure
    clf();
    subplot(2, 1, 1);
    plot2d3(nx, xn); plot(nx, xn, "r.");
    xgrid();
    title("Original Signal x(n)");
    xlabel("n"); ylabel("x(n)");
    
    subplot(2, 1, 2);
    plot2d3(ny, yn); plot(ny, yn, "b.");
    xgrid();
    title("Delayed Signal y(n) = x(n - " + string(k) + ")");
    xlabel("n"); ylabel("y(n)");
endfunction

// ----------------------------------------------------------------------------
// Exercise 2: Advance Operation y(n) = x(n + k), k > 0

    yn = xn;

    nx = (1:length(xn)) - xorigin;
    ny = (1:length(yn)) - yorigin;
    
    // Graphical display in a single figure
    clf();
    subplot(2, 1, 1);

    xgrid();

    xlabel("n"); ylabel("x(n)");
    
    subplot(2, 1, 2);
    plot2d3(ny, yn); plot(ny, yn, "b.");
    xgrid();

    xlabel("n"); ylabel("y(n)");
endfunction

// ----------------------------------------------------------------------------
// Exercise 3: Folding Operation y(n) = x(-n)
// ----------------------------------------------------------------------------
function [yn, yorigin] = fold(xn, xorigin)
    yn = xn($:-1:1); // Vector reversal
    yorigin = length(xn) - xorigin + 1;
    
    // Determine discrete time axes
    nx = (1:length(xn)) - xorigin;
    ny = (1:length(yn)) - yorigin;
    
    // Graphical display in a single figure
    clf();
    subplot(2, 1, 1);
    plot2d3(nx, xn); plot(nx, xn, "r.");
    xgrid();
    title("Original Signal x(n)");
    xlabel("n"); ylabel("x(n)");
    
    subplot(2, 1, 2);
    plot2d3(ny, yn); plot(ny, yn, "b.");
    xgrid();
    title("Folded Signal y(n) = x(-n)");
    xlabel("n"); ylabel("y(n)");
endfunction

// ----------------------------------------------------------------------------
// Exercise 4: Addition Operation y(n) = x1(n) + x2(n)
// ----------------------------------------------------------------------------
function [yn, yorigin] = add(x1n, x1origin, x2n, x2origin)
    n1 = (1:length(x1n)) - x1origin;
    n2 = (1:length(x2n)) - x2origin;
    
    n_min = min(min(n1), min(n2));
    n_max = max(max(n1), max(n2));
    n = n_min:n_max;
    
    // Allocate zero-padded vectors
    y1 = zeros(1, length(n));
    y2 = zeros(1, length(n));
    
    // Assign samples into unified coordinate space
    y1(min(n1) - n_min + 1 : max(n1) - n_min + 1) = x1n;
    y2(min(n2) - n_min + 1 : max(n2) - n_min + 1) = x2n;
    
    yn = y1 + y2;
    yorigin = 1 - n_min;
    
    // 3-panel display in one figure
    clf();
    subplot(3, 1, 1);
    plot2d3(n1, x1n); plot(n1, x1n, "r."); xgrid();
    title("Signal x1(n)"); xlabel("n"); ylabel("x1(n)");
    
    subplot(3, 1, 2);
    plot2d3(n2, x2n); plot(n2, x2n, "g."); xgrid();
    title("Signal x2(n)"); xlabel("n"); ylabel("x2(n)");
    
    subplot(3, 1, 3);
    plot2d3(n, yn); plot(n, yn, "b."); xgrid();
    title("Sum Signal y(n) = x1(n) + x2(n)"); xlabel("n"); ylabel("y(n)");
endfunction

// ----------------------------------------------------------------------------
// Exercise 5: Multiplication Operation y(n) = x1(n) .* x2(n)
// ----------------------------------------------------------------------------
function [yn, yorigin] = multi(x1n, x1origin, x2n, x2origin)
    n1 = (1:length(x1n)) - x1origin;
    n2 = (1:length(x2n)) - x2origin;
    
    n_min = min(min(n1), min(n2));
    n_max = max(max(n1), max(n2));
    n = n_min:n_max;
    
    // Allocate zero-padded vectors
    y1 = zeros(1, length(n));
    y2 = zeros(1, length(n));
    
    y1(min(n1) - n_min + 1 : max(n1) - n_min + 1) = x1n;
    y2(min(n2) - n_min + 1 : max(n2) - n_min + 1) = x2n;
    
    // Element-wise product
    yn = y1 .* y2;
    yorigin = 1 - n_min;
    
    // 3-panel display in one figure
    clf();
    subplot(3, 1, 1);
    plot2d3(n1, x1n); plot(n1, x1n, "r."); xgrid();
    title("Signal x1(n)"); xlabel("n"); ylabel("x1(n)");
    
    subplot(3, 1, 2);
    plot2d3(n2, x2n); plot(n2, x2n, "g."); xgrid();
    title("Signal x2(n)"); xlabel("n"); ylabel("x2(n)");
    
    subplot(3, 1, 3);
    plot2d3(n, yn); plot(n, yn, "b."); xgrid();
    title("Product Signal y(n) = x1(n) .* x2(n)"); xlabel("n"); ylabel("y(n)");
endfunction

// ----------------------------------------------------------------------------
// Exercise 6: Linear Convolution y(n) = x(n) * h(n)
// ----------------------------------------------------------------------------
function [yn, yorigin] = convolution(xn, xorigin, hn, horigin)
    // Perform discrete polynomial convolution
    yn = convol(xn, hn);
    yorigin = xorigin + horigin - 1;
    
    // Determine discrete time axes
    nx = (1:length(xn)) - xorigin;
    nh = (1:length(hn)) - horigin;
    ny = (1:length(yn)) - yorigin;
    
    // 3-panel display in one figure
    clf();
    subplot(3, 1, 1);
    plot2d3(nx, xn); plot(nx, xn, "r."); xgrid();
    title("Input Signal x(n)"); xlabel("n"); ylabel("x(n)");
    
    subplot(3, 1, 2);
    plot2d3(nh, hn); plot(nh, hn, "g."); xgrid();
    title("Impulse Response h(n)"); xlabel("n"); ylabel("h(n)");
    
    subplot(3, 1, 3);
    plot2d3(ny, yn); plot(ny, yn, "b."); xgrid();
    title("Convolution Result y(n) = x(n) * h(n)"); xlabel("n"); ylabel("y(n)");
endfunction

function y = x_sig(n)
    y = zeros(size(n));
    for i = 1:length(n)
        if (n(i) >= -3 & n(i) <= -1) then
            y(i) = 1 + n(i)/3;
        elseif (n(i) >= 0 & n(i) <= 3) then
            y(i) = 1;
        else
            y(i) = 0;
        end
    end
endfunction

// ----------------------------------------------------------------------------
// Exercise 2.2: Basic Signal Transformations
// Definition from Figure P2.2: x(n) = 1 for n in {-1, 0, 1, 2}, 0.5 for n in {3, 4}
// ----------------------------------------------------------------------------
clf();

function y = x_base(n)
    y = zeros(size(n));
    for i = 1:length(n)
        if (n(i) >= -1 & n(i) <= 2) then
            y(i) = 1;
        elseif (n(i) == 3 | n(i) == 4) then
            y(i) = 0.5;
        else
            y(i) = 0;
        end
    end
endfunction

n = -6:8;

ya = x_base(n - 2);
yb = x_base(4 - n);
yc = x_base(n + 2);
yd = x_base(n) .* bool2s(n <= 2);
ye = x_base(n - 1) .* bool2s(n == 3);
yf = x_base(n.^2);
yg = 0.5 * (x_base(n) + x_base(-n));
yh = 0.5 * (x_base(n) - x_base(-n));

signals = list(ya, yb, yc, yd, ye, yf, yg, yh);
titles = ["(a) x(n-2)", "(b) x(4-n)", "(c) x(n+2)", "(d) x(n)u(2-n)", ...
          "(e) x(n-1)delta(n-3)", "(f) x(n^2)", "(g) Even part", "(h) Odd part"];

for k = 1:8
    subplot(4, 2, k);
    plot2d3(n, signals(k)); plot(n, signals(k), "ro");
    xtitle(titles(k), "n", "Amplitude"); xgrid();
end

// ----------------------------------------------------------------------------
// Exercise 2.6: System Time-Invariance Analysis
// ----------------------------------------------------------------------------
clf();

function y = x_box(n_vec)
    y = zeros(size(n_vec));
    for i = 1:length(n_vec)
        if (n_vec(i) >= 0 & n_vec(i) <= 3) then
            y(i) = 1;
        else
            y(i) = 0;
        end
    end
endfunction

n_26 = -4:7;

// (b) y(n) = x(n^2)
yb_delayed_out = x_box((n_26 - 2).^2);
yb_response_in = x_box((n_26.^2) - 2);

// (c) y(n) = x(n) - x(n-1)
yc_delayed_out = x_box(n_26 - 2) - x_box(n_26 - 3);
yc_response_in = x_box(n_26 - 2) - x_box(n_26 - 3);

// (d) y(n) = n*x(n)
yd_delayed_out = (n_26 - 2) .* x_box(n_26 - 2);
yd_response_in = n_26 .* x_box(n_26 - 2);

signals_26 = list(yb_delayed_out, yb_response_in, ...
                  yc_delayed_out, yc_response_in, ...
                  yd_delayed_out, yd_response_in);

titles_26 = [
    "System (b): y(n-2)", ...
    "System (b): y_2(n) = T[x(n-2)] (Time-Variant)", ...
    "System (c): y(n-2)", ...
    "System (c): y_2(n) = T[x(n-2)] (Time-Invariant)", ...
    "System (d): y(n-2)", ...
    "System (d): y_2(n) = T[x(n-2)] (Time-Variant)"
];

y_limits = [

];

for k = 1:6
